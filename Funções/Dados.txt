#include <iostream>
#include <cstdlib>
#include <ctime>

using namespace std;

int soma3Dados(int d1, int d2, int d3)
{
    return d1 + d2 + d3;
}

int lanca3dados(int &d1, int &d2, int &d3)
{
    d1 = rand() % 6 + 1;
    d2 = rand() % 6 + 1;
    d3 = rand() % 6 + 1;

    return soma3Dados(d1, d2, d3);
}

int main()
{
    srand(time(0));

    int d1,d2,d3;
    int soma, melhorPontuacao=0, pontuacaoAnterior=0, jogadas=0;
    char resposta;

    while (jogadas < 8){
        cout << "gostaria de lancar os dados (S/N)? ";
        cin >> resposta;

        if (resposta == 'N' || resposta == 'n'){
            break;
        }

        cout << "a sua melhor pontuacao ate agora é " << melhorPontuacao << endl;
        soma = lanca3dados(d1, d2, d3);

        cout << "d1 = " << d1<< " d2 = " << d2<< " d3 = " << d3<< " ==> " << soma<< " pontos" << endl;

        if (jogadas > 0) {
            if (soma > pontuacaoAnterior){
                cout << "parabens! voce ganhou pontos em relacao a jogada anterior :) " << endl;
            }
            else if (soma < pontuacaoAnterior){
                cout << "que pena, voce perdeu pontos em relacao a jogada anterior :(" << endl;
            }
            else{
                cout << "Voce manteve a mesma pontuacao da jogada anterior" << endl;
            }
        }

        if (soma > melhorPontuacao){
            melhorPontuacao = soma;
        }

        pontuacaoAnterior = soma;
        jogadas++;
    }

    cout << "voce jogou " << jogadas << " vezes" << endl;

    return 0;
}
