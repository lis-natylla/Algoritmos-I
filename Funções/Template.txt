#include <iostream>
using namespace std;

template <typename T>
T media(T x, T y, T z){
    return (x+y+z)/3;
}

int main(){
    int i1, i2, i3;
    float f1, f2, f3;

    cout << "digite 3 numeros inteiros: ";
    cin >> i1 >> i2 >> i3;

    cout << "digite 3 numeros reais: ";
    cin >> f1 >> f2 >> f3;

    cout << "media dos inteiros: "<< media(i1, i2, i3) << endl;
    cout << "media dos reais: "<< media(f1, f2, f3) << endl;

    return 0;
}
