#include <iostream>

int ehprimo(int m)
{
    if (m <= 1){
        return 0;
    }

    if (m == 2){
        return 1;
    }

    if (m % 2 == 0){
        return 0;
    }

    for (int i = 3; i * i <= m; i += 2){
        if (m % i == 0){
            return 0;
        }
    }

    return 1;
}

using namespace std;

int main(){
    int numero;
    cin >> numero;

    cout << "resultado: " << ehprimo(numero) << endl;

    return 0;
}
