#include <iostream>
#include <cstdlib>
#include <ctime>
#include <iomanip>

using namespace std;

int main() {

    int matrizA[6][6];
    int matrizB[6][6];
    int Soma[6][6];

    srand(time(0));

    for (int i = 0; i < 6; i++) {
        for (int j = 0; j < 6; j++) {
            
            matrizA[i][j] = (i + 1) * (j + 1);

            matrizB[i][j] = rand() % 10;

            Soma[i][j] = matrizA[i][j] + matrizB[i][j];
        }
    }
    cout << "Matriz A:\n";
    for (int i = 0; i < 6; i++) {
        for (int j = 0; j < 6; j++) {
            cout << left<< setw(5)<< matrizA[i][j];
        }
        cout << endl;
    }
    
    cout << "\nMatriz B:\n";
    for (int i = 0; i < 6; i++) {
        for (int j = 0; j < 6; j++) {
            cout << left<< setw(5)<< matrizB[i][j];
        }
        cout << endl;
    }
    
    cout << "\nMatriz Soma:\n";
    for (int i = 0; i < 6; i++) {
        for (int j = 0; j < 6; j++) {
            cout << left<< setw(5)<< Soma[i][j];
        }
        cout << endl;
    }

    return 0;
}
